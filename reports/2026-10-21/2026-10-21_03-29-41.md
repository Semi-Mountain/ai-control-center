3. 任何与"盘前/盘中快照"正交的任务（周报、月报、专题）。

**最终范围 = A3'(9 份: 09:30 + 8 盘中) + A1(2 份: 09:25 / 15:00)**，共 11 个交易时段 job。

def _jobs() -> dict:
    return {j["id"]: j for j in json.loads(JOBS.read_text(encoding="utf-8"))["jobs"]}


def slot_from_schedule(expr: str) -> str | None:
    """从 cron expr 反解 HH:MM（仅支持 'M H * * 1-5'）。"""
    m = re.match(r"^(\d{1,2})\s+(\d{1,2})\s+\*\s+\*\s+1-5$", expr.strip())
    return f"{int(m.group(2)):02d}:{int(m.group(1)):02d}" if m else None


def schedule_check() -> list[str]:
    """双源交叉校验：jobs.json schedule ↔ 变量表 slot key。任一不符 → 报错。"""
    jobs, errs = _jobs(), []
    for slot, v in SLOT_TABLE.items():
        j = jobs.get(v["job_id"])
        if not j:
            errs.append(f"{slot}: job_id {v['job_id']} 不在 jobs.json"); continue
        derived = slot_from_schedule(j.get("schedule", {}).get("expr", ""))
        if derived != slot:
            errs.append(f"{slot}: schedule 反解={derived} ≠ 变量表 slot={slot} ({j['name']})")
        if j.get("name") != v["job_name"]:
            errs.append(f"{slot}: job_name 不符 jobs.json={j.get('name')}!={v['job_name']}")
    for slot, jid in BESPOKE.items():
        j = jobs.get(jid)
        if not j:
            errs.append(f"{slot}: bespoke job {jid} 缺失")
    return errs


def render(slot: str) -> str:
    if slot not in SLOT_TABLE:
        raise KeyError(f"slot {slot} 不在模板族（09:25/15:00 为 bespoke，保持手写）")
    errs = schedule_check()
    if errs:
        raise RuntimeError("[FAIL-CLOSED] schedule×变量表 交叉校验失败:\n  " + "\n  ".join(errs))
    v = dict(SLOT_TABLE[slot]); v["slot"] = slot; v["template_id"] = TEMPLATE_ID
    return Template(PROMPT_TEMPLATE, undefined=StrictUndefined).render(**v)


def self_check(text: str, slot: str) -> list[str]:
    """输出级章节自检：禁词只作用于 '## ' 章节标题，phase 自身豁免。"""
    errs, v = [], SLOT_TABLE[slot]
    titles = [l[3:].strip() for l in text.splitlines() if l.startswith("## ")]
    phase = v["session_phase"]
    for t in titles:
        for bad in v["forbidden_chapter_keywords"]:
            if bad in t and bad not in phase:
                errs.append(f"[{slot}] 章节标题含禁词 '{bad}': {t}")
    for need in (v["news_chapter"], v["holdings_chapter"], v["market_chapter"]):
        if not any(need in t for t in titles):
            errs.append(f"[{slot}] 缺少必需章节 '{need}'")
    return errs


def main() -> int:
    cmd = sys.argv[1] if len(sys.argv) > 1 else "check"
    if cmd == "render":
        print(render(sys.argv[2])); return 0
    if cmd == "schedule-check":
        errs = schedule_check()
        print("\n".join(errs) if errs else "[OK] schedule × 变量表 交叉校验全部通过")
        return 1 if errs else 0
    bad = []
    for slot in SLOT_TABLE:
        try:
            bad += self_check(render(slot), slot)
        except Exception as e:
            bad.append(f"[{slot}] render 异常: {e}")
    print(f"渲染槽位: {len(SLOT_TABLE)}（bespoke 手写: {', '.join(BESPOKE)}）")
    print("\n".join(bad) if bad else "[OK] 9 槽位渲染 + 章节自检全部通过")
    return 1 if bad else 0


if __name__ == "__main__":
    raise SystemExit(main())
